# Privacy Policy Disclosures Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** When users scaffold analytics tools, the CLI writes matching privacy disclosures into the project's privacy policy (creating `PRIVACY.md` if none exists), plus a standalone `privacy` command for existing installations.

**Architecture:** A new `src/privacy.js` module holds pure text-transform helpers (strip-blocks, managed-block upsert) and an IO wrapper `patchPrivacyPolicy(cwd, tools)`. Two entry points call it: a new default-yes toggle in `init`, and a new `analytics-stack privacy` subcommand that detects installed tools from `.env.example`. `verify` gains a soft warning. Disclosure content lives in `templates/privacy/` using the same BEGIN/END strip-block idiom as `templates/analytics/analytics.js`.

**Tech Stack:** Node 18+ ESM, `prompts`, `picocolors`, built-in `node:test` (no new dependencies).

**Spec:** `docs/superpowers/specs/2026-06-06-privacy-policy-disclosures-design.md`

```js
// Privacy policy generation — shared by `init` and the `privacy` command.
import { existsSync, readFileSync } from "node:fs";
import { join } from "node:path";
import { TEMPLATES, readText, writeText, warn } from "./utils.js";
import { envPrefix } from "./detect.js";

export const BLOCK_BEGIN = "<!-- BEGIN analytics-stack -->";
export const BLOCK_END = "<!-- END analytics-stack -->";

// First match wins. On case-insensitive filesystems (macOS) some of these alias.
export const POLICY_FILENAMES = ["PRIVACY.md", "privacy.md", "privacy-policy.md", "PRIVACY-POLICY.md"];

const TOOL_MARKERS = { ga4: "GA4", clarity: "CLARITY", posthog: "POSTHOG" };

// Strip a guarded block like <!-- BEGIN GA4 --> ... <!-- END GA4 --> when the
// tool is unselected; drop just the marker lines when it is selected.
// Same idiom as stripBlock() in init.js, adapted to HTML comments.
export function stripToolBlocks(text, tools) {
  let out = text;
  for (const [tool, name] of Object.entries(TOOL_MARKERS)) {
    if (tools.has(tool)) {
      out = out.replace(new RegExp(`[ \\t]*<!-- BEGIN ${name} -->\\n?`, "g"), "");
      out = out.replace(new RegExp(`[ \\t]*<!-- END ${name} -->\\n?`, "g"), "");
    } else {
      out = out.replace(
        new RegExp(`\\n?[ \\t]*<!-- BEGIN ${name} -->[\\s\\S]*?<!-- END ${name} -->\\n?`, "g"),
        "\n"
      );
    }
  }
  return out.replace(/\n{3,}/g, "\n\n");
}

function escapeRe(s) {
  return s.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
}

// Replace the managed block if present, append if absent. Never touches text
// outside the markers. Broken markers (BEGIN without END or vice versa) get a
// fresh block appended instead — repairing would risk eating user content.
export function upsertManagedBlock(policyText, block) {
  const hasBegin = policyText.includes(BLOCK_BEGIN);
  const hasEnd = policyText.includes(BLOCK_END);
  if (hasBegin && hasEnd) {
    const re = new RegExp(`${escapeRe(BLOCK_BEGIN)}[\\s\\S]*?${escapeRe(BLOCK_END)}`);
    return { content: policyText.replace(re, block.trim()), action: "replaced" };
  }
  const appended = policyText.trimEnd() + "\n\n" + block.trim() + "\n";
  return { content: appended, action: hasBegin || hasEnd ? "appended-broken" : "appended" };
}
```

```js
// Build the full managed block for the selected tools.
export function buildPrivacySection(tools) {
  const template = readFileSync(join(TEMPLATES, "privacy", "privacy-section.md"), "utf8");
  const section = stripToolBlocks(template, tools).trim();
  return `${BLOCK_BEGIN}\n${section}\n${BLOCK_END}`;
}

export function findPolicyFile(cwd) {
  for (const name of POLICY_FILENAMES) {
    const path = join(cwd, name);
    if (existsSync(path)) return path;
  }
  return null;
}

// Create or update the project's privacy policy. Returns { path, created },
// or null when the existing file could not be read (warned, skipped).
export function patchPrivacyPolicy(cwd, tools) {
  const block = buildPrivacySection(tools);
  const existingPath = findPolicyFile(cwd);

  if (!existingPath) {
    const skeleton = readFileSync(join(TEMPLATES, "privacy", "privacy-skeleton.md"), "utf8");
    const today = new Date().toISOString().slice(0, 10);
    const content = skeleton.replace("__DATE__", today).replace("__ANALYTICS_SECTION__", block);
    const path = join(cwd, "PRIVACY.md");
    writeText(path, content);
    return { path, created: true };
  }

  const existing = readText(existingPath);
  if (existing === null) {
    warn(`Could not read ${existingPath} — skipping privacy policy update.`);
    return null;
  }
  const { content, action } = upsertManagedBlock(existing, block);
  if (action === "appended-broken") {
    warn(
      "Found a broken analytics-stack marker block in your privacy policy — appended a fresh block instead. Please remove the old one manually."
    );
  }
  writeText(existingPath, content);
  return { path: existingPath, created: false };
}
```

```js
// Detect previously-installed tools from .env.example (same keys verify checks).
// Returns a Set of tool ids, or null when nothing was detected.
export function detectToolsFromEnvExample(cwd, framework) {
  const content = readText(join(cwd, ".env.example"));
  if (!content) return null;
  const prefix = envPrefix(framework);
  const tools = new Set();
  if (content.includes(`${prefix}GA4_ID`)) tools.add("ga4");
  if (content.includes(`${prefix}CLARITY_ID`)) tools.add("clarity");
  if (content.includes(`${prefix}POSTHOG_KEY`)) tools.add("posthog");
  return tools.size > 0 ? tools : null;
}
```

```js
// `privacy` command — add/update privacy policy disclosures for an existing
// installation, without touching any other scaffolded file.
import prompts from "prompts";
import { c, header, step, info, blank, error } from "../utils.js";
import { detectFramework, getFrameworks, getFrameworkById } from "../detect.js";
import { patchPrivacyPolicy, detectToolsFromEnvExample } from "../privacy.js";

export async function privacyCommand() {
  const cwd = process.cwd();
  process.stdout.write(`\n${header("analytics-stack")} ${c.dim("privacy")}\n\n`);
  info(`Working directory: ${c.cyan(cwd)}`);

  let framework = detectFramework(cwd);
  if (framework) {
    info(`Detected framework: ${c.bold(framework.label)} ${c.dim(`(${framework.id})`)}`);
  } else {
    const { frameworkId } = await prompts(
      {
        type: "select",
        name: "frameworkId",
        message: "Which framework is this project?",
        choices: getFrameworks().map((fw) => ({ title: fw.label, value: fw.id })),
      },
      { onCancel: () => process.exit(1) }
    );
    framework = getFrameworkById(frameworkId);
    if (!framework) {
      error("Aborted.");
      process.exit(1);
    }
  }

  const detected = detectToolsFromEnvExample(cwd, framework);
  if (detected) {
    info(`Detected tools in .env.example: ${c.bold([...detected].join(", "))}`);
  }
  blank();

  const { tools } = await prompts(
    {
      type: "multiselect",
      name: "tools",
      message: "Which analytics tools should the policy disclose?",
      instructions: false,
      hint: "- Space to toggle, Enter to confirm",
      choices: [
        { title: "Google Analytics 4", value: "ga4", selected: !detected || detected.has("ga4") },
        { title: "Microsoft Clarity", value: "clarity", selected: !detected || detected.has("clarity") },
        { title: "PostHog", value: "posthog", selected: !detected || detected.has("posthog") },
      ],
      min: 1,
    },
    { onCancel: () => process.exit(1) }
  );

  const result = patchPrivacyPolicy(cwd, new Set(tools));
  if (!result) {
    error("Privacy policy was not updated.");
    process.exit(1);
  }
  blank();
  step(result.created ? "Created" : "Updated", relPath(cwd, result.path));
  blank();
  process.stdout.write(
    `${c.bold(c.green("✨ Done!"))} Review the file, replace the ${c.cyan("[Your Company]")} placeholders, and link it from your site footer.\n\n`
  );
}

function relPath(cwd, abs) {
  return abs.startsWith(cwd) ? abs.slice(cwd.length + 1) : abs;
```

````markdown
# Privacy policy generation

Analytics vendors require you to disclose data collection to your visitors.
GA4's terms include a "User Data Collection Acknowledgement"; Clarity and
PostHog have equivalent expectations. `analytics-stack` generates the matching
disclosure text for exactly the tools you installed.

> ⚠️ **Not legal advice.** The generated text is a starting point. Review it
> with a qualified professional and adapt it to your jurisdiction before
> publishing.

## How it works

Two entry points, same engine:

- **`npx analytics-stack init`** — asks "Add analytics disclosures to your
  privacy policy?" (default: yes) as part of scaffolding.
- **`npx analytics-stack privacy`** — standalone command for existing
  installations. Detects your framework and which tools you installed (from
  `.env.example`), pre-selects them, and updates only the policy file. Nothing
  else in your project is touched.

## What gets written

The CLI looks for an existing policy file at the project root, in this order:
`PRIVACY.md`, `privacy.md`, `privacy-policy.md`, `PRIVACY-POLICY.md`.

**If none exists**, it creates `PRIVACY.md` from a skeleton: title, last-updated
date, `[Your Company]` / `[Contact Email]` placeholders, a not-legal-advice
note, and the disclosure block.

**If one exists**, only the managed block is added or replaced:

```markdown
<!-- BEGIN analytics-stack -->
## Analytics & Cookies
...one section per installed tool...
<!-- END analytics-stack -->
```

## The marker-block contract

- Text **outside** the markers is yours — the CLI never modifies it.
- Text **inside** the markers belongs to the CLI — it is fully replaced on every
  run, so don't hand-edit inside the block (your edits would be overwritten).
  Want different wording? Move the text outside the markers and delete the block.
- Re-running after changing your tool selection updates the block to match.
- If the markers are broken (a `BEGIN` without an `END`), the CLI appends a
  fresh block and warns you to remove the old one manually — it never guesses
  at deleting your content.

## Per-tool disclosures

Each enabled tool gets a section covering: who the vendor is, what data it
collects, a link to the vendor's privacy policy, an opt-out link where the
vendor provides one, and a note that nothing is collected before the visitor
accepts analytics cookies (which is literally true — see
[consent gating](./consent-mode.md)).

| Tool | Vendor policy | Opt-out |
|---|---|---|
| Google Analytics 4 | https://policies.google.com/privacy | https://tools.google.com/dlpage/gaoptout |
| Microsoft Clarity | https://privacy.microsoft.com/privacystatement | — |
| PostHog | https://posthog.com/privacy | — |

## After generating

1. Replace `[Your Company]` and `[Contact Email]` placeholders.
2. Review the text (and ideally have a professional review it).
3. Link the policy from your site footer.
4. `npx analytics-stack verify` will warn (softly — it never fails the run) if
   your policy is missing the disclosure block.
````

```js
// End-to-end smoke: init + privacy + verify in a temp project.
// Run from the repo root: node scripts/smoke-e2e.mjs
import { mkdtempSync, writeFileSync, readFileSync, rmSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { spawnSync } from "node:child_process";
import prompts from "prompts";

const REPO = process.cwd();
const CLI = join(REPO, "bin", "cli.js");
const tmp = mkdtempSync(join(tmpdir(), "astack-smoke-"));

writeFileSync(
  join(tmp, "package.json"),
  JSON.stringify({ name: "smoke-app", dependencies: { vite: "^5.0.0", react: "^18.0.0" } }, null, 2)
);
process.chdir(tmp);

function fail(msg) {
  console.error(`✖ SMOKE FAIL: ${msg} (dir kept for inspection: ${tmp})`);
  process.exit(1);
}

// --- 1. init with privacy enabled, GA4 + Clarity only, no dep install ---
prompts.inject([
  "vite-react",       // framework
  ["ga4", "clarity"], // tools
  false,              // typescript
  "src",              // outDir
  true,               // privacy
  false,              // installDeps
  true,               // confirm
]);
const { initCommand } = await import(join(REPO, "src", "commands", "init.js"));
await initCommand([]);

let policy = readFileSync(join(tmp, "PRIVACY.md"), "utf8");
if (!policy.includes("Google Analytics 4")) fail("init: GA4 section missing");
if (!policy.includes("Microsoft Clarity")) fail("init: Clarity section missing");
if (policy.includes("PostHog")) fail("init: PostHog section should be absent");
if (!policy.includes("[Your Company]")) fail("init: skeleton placeholders missing");

// --- 2. privacy re-run with all three tools — block must be replaced, once ---
prompts.inject([["ga4", "clarity", "posthog"]]);
const { privacyCommand } = await import(join(REPO, "src", "commands", "privacy.js"));
await privacyCommand([]);

policy = readFileSync(join(tmp, "PRIVACY.md"), "utf8");
if (!policy.includes("PostHog")) fail("privacy: block not updated with PostHog");
if (policy.split("<!-- BEGIN analytics-stack -->").length !== 2) fail("privacy: duplicate managed blocks");

// --- 3. verify — no privacy warning when the block exists ---
let res = spawnSync(process.execPath, [CLI, "verify"], { cwd: tmp, encoding: "utf8" });
let out = res.stdout + res.stderr;
if (out.includes("No privacy policy found")) fail("verify: warned despite policy existing");
if (out.includes("has no analytics disclosures")) fail("verify: warned despite block existing");

// --- 4. verify — warning appears when the policy is gone ---
rmSync(join(tmp, "PRIVACY.md"));
res = spawnSync(process.execPath, [CLI, "verify"], { cwd: tmp, encoding: "utf8" });
out = res.stdout + res.stderr;
if (!out.includes("No privacy policy found")) fail("verify: missing-policy warning not shown");

console.log(`\n✨ smoke OK (${tmp})`);
rmSync(tmp, { recursive: true, force: true });
```
